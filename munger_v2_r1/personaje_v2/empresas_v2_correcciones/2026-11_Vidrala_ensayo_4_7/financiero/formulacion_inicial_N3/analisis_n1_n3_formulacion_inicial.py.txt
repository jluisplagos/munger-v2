def paginas(f):
    if f not in _cache:
        t = open(os.path.join(TXT, f), encoding="utf-8").read()
        partes = re.split(r"\[PDF PAGE (\d+)\]", t)

    return _cache[f]


def numeros(linea):
    out = []
    for m in re.finditer(r"\(?-?\d{1,3}(?:[,.]\d{3})+\)?|\(?-?\d+\)?", linea):
        s = m.group(0)
        neg = s.startswith("(") or s.startswith("-")
        v = float(re.sub(r"[(),.\-]", "", s))
        out.append(-v if neg else v)
    return out

def ev(clave, f, patron, col, esperado_miles, concepto, periodo, perimetro, signo="importe con el signo del estado", pagina=None, nota="", ocurrencia=1):
    """Busca la primera línea que casa con `patron` (en la página dada, si se da) y toma el número `col` de los que siguen al
    patrón (0 = primera columna numérica, normalmente el año del informe; 1 = comparativo)."""
    for pag, tx in paginas(f):
        if pagina and pag != pagina:
            continue

            if m:

                if vistas < ocurrencia:

                ns = [x for x in ns if not (len(ns) > 2 and abs(x) < 40 and x == int(x) and ns.index(x) == 0)]  # quita el nº de nota
                if len(ns) > col:
                    impresa = [x for x in re.findall(r"^\s*(\d{1,3})\s*$", tx, flags=re.M)]
                    rec = {"clave": clave, "archivo": f, "pagina_pdf": pag, "pagina_impresa": int(impresa[-1]) if impresa else None, "linea": l.strip()[:160],
                           "concepto": concepto, "columna": col, "periodo": periodo, "perimetro": perimetro, "unidad": "miles de euros", "signo": signo,
                           "valor_miles": ns[col], "esperado_miles": esperado_miles, "reproduce": abs(abs(ns[col]) - abs(esperado_miles)) < 0.5, "nota": nota}
                    E.append(rec)
                    return abs(esperado_miles) / 1000.0
    E.append({"clave": clave, "archivo": f, "reproduce": False, "nota": "no encontrado"})

def proyeccion(t, m0, regla, esc):
    """Año a año (1–10): ventas, margen, NOPAT, reinversión neta, capital liberado y FCF. regla: 'simetrica' (la de la plantilla:
    reinversión = ΔV/S2C también si ΔV < 0), 'circulante' (con ventas que caen solo se libera el circulante comercial:
    ΔV × NWC/ventas; la inversión fija no se recupera) o 'ninguna' (con ventas que caen no se libera capital)."""
    g1, m5, s2c, g, roic = esc
    V = [V0]
    filas = []
    for k in range(1, 11):
        gk = g1 if k <= 5 else g1 + (g - g1) * (k - 5) / 5
        V.append(V[-1] * (1 + gk))
        mk = m0 + (m5 - m0) * min(k, 5) / 5
        nopat = V[k] * mk * (1 - t)
        dv = V[k] - V[k - 1]
        if dv >= 0 or regla == "simetrica":
            rein = dv / s2c
        elif regla == "circulante":
            rein = dv * NWC_RATIO
        else:
            rein = 0.0
        filas.append({"anio": k, "ventas": V[k], "margen": mk, "nopat": nopat, "reinversion": rein, "capital_liberado": max(0.0, -rein), "fcf": nopat - rein})
    return V, filas


def valor(t, m0, regla, esc, r=None):
    r = r_op(t) if r is None else r
    g1, m5, s2c, g, roic = esc
    V, fil = proyeccion(t, m0, regla, esc)
    pv = sum(f["fcf"] / (1 + r) ** f["anio"] for f in fil)
    n11 = V[10] * (1 + g) * m5 * (1 - t)
    tv = n11 * (1 - g / roic) / (r - g)
    pvt = tv / (1 + r) ** 10
    return (pv + pvt - PUENTE) / ACC, pvt / (pv + pvt), fil


def escalera(Vs, p, K=3.0):
    lo, hi = 0.0, max(Vs)
    for _ in range(200):
        mid = (lo + hi) / 2
        G = sum(pi * max(v - mid, 0) for v, pi in zip(Vs, p))
        L = sum(pi * max(mid - v, 0) for v, pi in zip(Vs, p))
        lo, hi = (mid, hi) if G >= K * L else (lo, mid)
    return min(lo, Vs[1]), sum(v * pi for v, pi in zip(Vs, p))
